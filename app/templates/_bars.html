{# The progress bar used across the screens. Kept in one place so the installments, subscriptions and summary pages cannot drift apart visually. #} {% macro meter(percent, extra='') -%} {%- set pct = [[percent | round | int, 0] | max, 100] | min -%} {%- endmacro %} {# A bar plus "3/12 paid (25%)", for an installment plan or a bank. #} {% macro progress(done, total, extra='', noun='paid') -%} {%- set pct = (done * 100 / total) if total else 0 -%} {{ meter(pct, extra) }}{{ done }}/{{ total }} {{- ' ' }}{{ noun }} ({{ pct | round | int }}%) {%- endmacro %} {# A bar plus "R$ 500,00 of R$ 850,00", when the progress is money rather than a count of installments. The caller passes the already-formatted amounts. #} {% macro amounts(part, whole, part_text, whole_text, extra='') -%} {%- set pct = (part * 100 / whole) if whole else 0 -%} {{ meter(pct, extra) }}{{ part_text }} of {{- ' ' }}{{ whole_text }} ({{ pct | round | int }}%) {%- endmacro %} {# A bar plus a bare percentage, for "share of the month" columns. #} {% macro share(part, whole, extra='') -%} {%- if whole and whole > 0 -%} {%- set pct = part * 100 / whole -%} {{ meter(pct, extra) }}{{ pct | round | int }}% {%- endif -%} {%- endmacro %}
